
cmake_minimum_required(VERSION 3.1...4.0)
project(moonlight-common-c LANGUAGES C)

set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

option(USE_MBEDTLS "Use MbedTLS instead of OpenSSL" OFF)
option(CODE_ANALYSIS "Run code analysis during compilation" OFF)

SET(CMAKE_C_STANDARD 11)

set(CMAKE_POSITION_INDEPENDENT_CODE_BACKUP ${CMAKE_POSITION_INDEPENDENT_CODE})
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(enet)
set(CMAKE_POSITION_INDEPENDENT_CODE ${CMAKE_POSITION_INDEPENDENT_CODE_BACKUP})
unset(CMAKE_POSITION_INDEPENDENT_CODE_BACKUP)

aux_source_directory(src SRC_LIST)
aux_source_directory(reedsolomon SRC_LIST)

# Build shared library by default, but allows user override
if (NOT DEFINED BUILD_SHARED_LIBS)
  set(BUILD_SHARED_LIBS ON)
  set(BUILD_SHARED_LIBS_OVERRIDE ON)
endif()

add_library(moonlight-common-c ${SRC_LIST})

if (BUILD_SHARED_LIBS_OVERRIDE)
  unset(BUILD_SHARED_LIBS)
  unset(BUILD_SHARED_LIBS_OVERRIDE)
endif()

target_link_libraries(moonlight-common-c PRIVATE enet)

if(MSVC)
  target_compile_options(moonlight-common-c PRIVATE /W3 /wd4100 /wd4232 /wd5105 /WX)
  target_link_libraries(moonlight-common-c PRIVATE ws2_32.lib winmm.lib)
elseif(MINGW)
  target_link_libraries(moonlight-common-c PRIVATE -lws2_32 -lwinmm)
else()
  target_compile_options(moonlight-common-c PRIVATE -Wall -Wextra -Wno-unused-parameter -Werror)
  if (CODE_ANALYSIS AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(moonlight-common-c PRIVATE -fanalyzer)
  endif()
endif()

if (USE_MBEDTLS)
  target_compile_definitions(moonlight-common-c PRIVATE USE_MBEDTLS)
  find_package(MbedTLS QUIET)
  if (MBEDTLS_FOUND)
    target_link_libraries(moonlight-common-c PRIVATE ${MBEDCRYPTO_LIBRARY})
    target_include_directories(moonlight-common-c SYSTEM PRIVATE ${MBEDTLS_INCLUDE_DIRS})
  else()
    # For sub project added via CMake
    target_link_libraries(moonlight-common-c PRIVATE mbedcrypto)
  endif()
else()
  find_package(OpenSSL 1.0.2 REQUIRED)
  target_link_libraries(moonlight-common-c PRIVATE ${OPENSSL_CRYPTO_LIBRARY})
  target_include_directories(moonlight-common-c SYSTEM PRIVATE ${OPENSSL_INCLUDE_DIR})
endif()

string(TOUPPER "x${CMAKE_BUILD_TYPE}" BUILD_TYPE)
if("${BUILD_TYPE}" STREQUAL "XDEBUG")
  target_compile_definitions(moonlight-common-c PRIVATE LC_DEBUG)
else()
  target_compile_definitions(moonlight-common-c PRIVATE NDEBUG)

  # Avoid false "maybe uninitialized" warning generated by old GCC versions
  # when building with -O2
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(moonlight-common-c PRIVATE -Wno-maybe-uninitialized)
  endif()
endif()

target_include_directories(moonlight-common-c SYSTEM PUBLIC src)

target_include_directories(moonlight-common-c PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/reedsolomon
)

target_compile_definitions(moonlight-common-c PRIVATE HAS_SOCKLEN_T)

option(LC_BUILD_INPUT_STREAM_TESTS "Build input stream ordering tests" OFF)
if(LC_BUILD_INPUT_STREAM_TESTS)
  enable_testing()
  target_compile_definitions(moonlight-common-c PRIVATE LC_INPUT_STREAM_TEST)
  if(WIN32)
    set_target_properties(moonlight-common-c PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  endif()
  add_executable(input-stream-order-test tests/InputStreamOrderTest.c)
  target_include_directories(input-stream-order-test PRIVATE tests)
  target_link_libraries(input-stream-order-test PRIVATE moonlight-common-c)
  add_test(NAME input-stream-order-test COMMAND input-stream-order-test)
endif()

option(LC_BUILD_CONNECTION_TERMINATION_TESTS "Build connection termination lifecycle tests" OFF)
if(LC_BUILD_CONNECTION_TERMINATION_TESTS)
  enable_testing()
  # The fixture includes production Connection.c with a controlled allocator and
  # supplies deterministic platform/transport stubs. No real connection is made.
  add_executable(connection-termination-test tests/ConnectionTerminationTest.c src/FakeCallbacks.c)
  target_include_directories(connection-termination-test PRIVATE src reedsolomon enet/include)
  target_compile_definitions(connection-termination-test PRIVATE HAS_SOCKLEN_T NDEBUG)
  add_test(NAME connection-termination-test COMMAND connection-termination-test)
endif()

option(LC_BUILD_AUTHORED_HAPTICS_TESTS "Build authored haptics parser and controller callback tests" OFF)
if(LC_BUILD_AUTHORED_HAPTICS_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(authored-haptics-test tests/AuthoredHapticsTest.c
    src/Ds5HapticsStream.c src/Ds5HapticsIrStream.c src/ByteBuffer.c
    src/LinkedBlockingQueue.c src/Platform.c)
  target_include_directories(authored-haptics-test PRIVATE src reedsolomon enet/include)
  target_compile_definitions(authored-haptics-test PRIVATE HAS_SOCKLEN_T LC_DEBUG)
  target_link_libraries(authored-haptics-test PRIVATE Threads::Threads)
  # The fixture calls production control dispatch without linking unrelated network APIs.
  if(MSVC)
    target_compile_options(authored-haptics-test PRIVATE /Gy /Gw /UNDEBUG)
    set_property(TARGET authored-haptics-test APPEND_STRING PROPERTY LINK_FLAGS " /OPT:REF")
  else()
    target_compile_options(authored-haptics-test PRIVATE -O2 -flto -ffunction-sections -fdata-sections -UNDEBUG)
    set_property(TARGET authored-haptics-test APPEND_STRING PROPERTY LINK_FLAGS " -flto")
    if(APPLE)
      set_property(TARGET authored-haptics-test APPEND_STRING PROPERTY LINK_FLAGS " -Wl,-dead_strip")
    else()
      set_property(TARGET authored-haptics-test APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--gc-sections")
    endif()
  endif()
  if(WIN32)
    target_link_libraries(authored-haptics-test PRIVATE ws2_32 winmm)
  endif()
  add_test(NAME authored-haptics-test COMMAND authored-haptics-test)
endif()
